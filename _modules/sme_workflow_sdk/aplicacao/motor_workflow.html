class MotorWorkflow:
    """Coordena a validação e a execução das transições do workflow.

    Recebe uma configuração previamente carregada e validada pelo
    :class:`ConfiguradorWorkflow
    <sme_workflow_sdk.configuracao.configurador.ConfiguradorWorkflow>`,
    converte seus estados e ações em entidades de domínio e utiliza os
    resolvedores para validar as decisões e as transições permitidas.

    O método :meth:`validar()` verifica se o estado e a ação informados são
    permitidos e se a transição correspondente respeita as regras
    de consistência do workflow.

    O método :meth:`executar()` produz o resultado da transição validada,
    contendo os estados envolvidos, a ação, os horários, o autor
    e o contexto da execução.

    O motor não executa regras de negócio específicas da aplicação
    consumidora, não altera o estado persistido da entidade e não
    realiza operações de persistência.

    Attributes:
        _configuracao: Configuração carregada e validada do workflow.
        _estados: Estados e ações convertidos em entidades de domínio.
        _decisao_resolver: Responsável por validar estados e ações.
        _transicao_resolver: Responsável por validar a consistência
            das transições.


    Exemples:
        >>> configurador = ConfiguradorWorkflow()
        >>> configuracao = configurador.carregar("workflow.yaml")
        >>> configurador.validar(configuracao)
        >>> motor = MotorWorkflow(configuracao)
        >>> execucao = motor.validar("pendente", "aprovar")
    """

    def __init__(self, configuracao: dict[str, Any]) -> None:
        """Inicializa o motor com uma configuração previamente validada.

        Converte os estados e as ações da configuração em entidades de
        domínio e inicializa os resolvedores responsáveis pela validação
        das decisões e das transições.

        A configuração é mantida em memória e não é relida do arquivo YAML.

        Args:
            configuracao (dict[str, Any]): Configuração do workflow carregada
            e validada pelo :class:`ConfiguradorWorkflow
            <sme_workflow_sdk.configuracao.configurador.ConfiguradorWorkflow>`.
        """
        self._configuracao = configuracao
        self._estados = self._converter_estados(configuracao)
        self._decisao_resolver = DecisaoResolver(self._estados)
        self._transicao_resolver = TransicaoResolver(self._estados)

    @staticmethod
    def _converter_estados(
        configuracao: dict[str, Any],
    ) -> dict[str, Estado]:
        """Converte os estados e suas ações em entidades de domínio.

        Percorre os estados definidos na configuração e cria instâncias
        de  :class:`Estado <sme_workflow_sdk.dominio.entidades.Estado>` e
        :class:`Acao <sme_workflow_sdk.dominio.entidades.Acao>` para
        representar as regras do workflow.

        Estados com ``acoes: null`` são representados com ``acoes=None``,
        indicando que não possuem ações disponíveis.

        Este método pressupõe que a configuração já passou pelas validações
        estruturais e semânticas do :class:`ConfiguradorWorkflow
        <sme_workflow_sdk.configuracao.configurador.ConfiguradorWorkflow>`.

        Args:
            configuracao (dict[str, Any]): Configuração validada contendo os
                estados e suas respectivas ações.

        Returns:
            dict[str, Estado]: Estados indexados por seus nomes.
        """
        estados: dict[str, Estado] = {}

        for nome_estado, dados_estado in configuracao["estados"].items():
            dados_acoes = dados_estado["acoes"]
            acoes: dict[str, Acao] | None = None

            if dados_acoes is not None:
                acoes = {
                    nome_acao: Acao(
                        nome=nome_acao,
                        proximo_estado=dados_acao["proximo_estado"],
                        terminal=dados_acao["terminal"],
                    )
                    for nome_acao, dados_acao in dados_acoes.items()
                }
            estados[nome_estado] = Estado(
                nome=nome_estado,
                acoes=acoes,
            )

        return estados

    def validar(self, estado_atual: str, acao: str) -> ExecucaoValidada:
        """Valida a ação solicitada e a transição correspondente.

        Verifica se o estado informado existe, se a ação está disponível
        para esse estado e se a transição configurada respeita as regras
        de consistência do workflow.

        A validação deve ocorrer antes de a aplicação consumidora executar
        as regras de negócio associadas à ação.

        Args:
            estado_atual (str): Nome do estado atual do workflow.
            acao (str): Nome da ação solicitada.

        Raises:
            EstadoInvalidoError: Se o estado não estiver configurado.
            AcaoInvalidaError: Se a ação não estiver disponível para o
                estado ou se os dados da transição forem inválidos.
            TransicaoInvalidaError: Se a transição configurada for inválida.

        Returns:
            ExecucaoValidada: Dados da execução com o estado atual, a ação
            configurada e o horário de início da transição.

        Examples:
            >>> execucao = motor.validar("pendente", "aprovar")
            >>> execucao.estado_atual
            'pendente'
        """
        logger.info(
            "Validando ação do workflow. Estado atual: %s | Ação: %s",
            estado_atual,
            acao,
        )
        try:
            execucao = self._decisao_resolver.validar(estado_atual, acao)
            transicao = self._transicao_resolver.resolver(execucao)

            logger.info(
                "Ação validada com sucesso. Estado atual: %s | "
                "Ação: %s | Próximo estado: %s",
                execucao.estado_atual,
                execucao.acao.nome,
                execucao.acao.proximo_estado,
            )

            return transicao

        except Exception:
            logger.exception(
                "Falha ao validar ação do workflow. "
                "Estado atual: %s | Ação: %s",
                estado_atual,
                acao,
            )
            raise

    def executar(
        self,
        dados_transicao: ExecucaoValidada,
        contexto: ContextoExecucao,
    ) -> dict[str, Any]:
        """Produz o resultado de uma execução do workflow.

        Recebe os dados previamente validados e o contexto da execução para
        montar o resultado da transição, incluindo os estados envolvidos, a
        ação, os horários, o autor e os dados de contexto.

        Deve ser chamado após a aplicação consumidora executar com sucesso as
        regras de negócio associadas à ação.

        Este método não executa regras de negócio, não altera o estado
        persistido da entidade e não realiza operações de persistência.

        Se ocorrer uma :class:`TransicaoInvalidaError
        <sme_workflow_sdk.excecoes.workflow.TransicaoInvalidaError>` durante a
        resolução da transição, o método captura a exceção e retorna um
        resultado indicando que a execução não foi bem-sucedida.

        Args:
            dados_transicao (ExecucaoValidada): Dados da transição
                previamente validados  método validar().
            contexto (ContextoExecucao): Contexto da execução, incluindo
                identificador, autor e dados adicionais.

        Returns:
            dict[str, Any]: Dicionário contendo os dados do resultado da
            execução, incluindo identificador, estado anterior, ação,
            próximo estado, terminalidade, indicador de sucesso, mensagem,
            horários, autor e contexto.

            Os valores de data e hora permanecem como objetos datetime,
            conforme o comportamento de :meth:`ResultadoWorkflow.para_dict()
            <sme_workflow_sdk.dominio.resultado.ResultadoWorkflow.para_dict>`

        Examples:
            >>> execucao = motor.validar("pendente", "aprovar")
            >>> resultado = motor.executar(execucao, contexto)
            >>> resultado["sucesso"]
            True
        """
        logger.info(
            "Iniciando execução do workflow. Identificador: %s | "
            "Estado atual: %s | Ação: %s | Autor: %s",
            contexto.identificador,
            dados_transicao.estado_atual,
            dados_transicao.acao.nome,
            contexto.autor,
        )
        try:
            execucao = self._transicao_resolver.resolver(dados_transicao)
            sucesso = True
            mensagem = "Transição executada com sucesso!"
            acao = execucao.acao
            resultado = ResultadoWorkflow(
                identificador=contexto.identificador,
                estado_anterior=execucao.estado_atual,
                acao=acao.nome,
                proximo_estado=acao.proximo_estado,
                terminal=acao.terminal,
                sucesso=sucesso,
                inicio=execucao.inicio_transicao,
                fim=datetime.now(UTC),
                autor=contexto.autor,
                contexto=contexto,
                mensagem=mensagem,
            )

            logger.info(
                "Workflow executado com sucesso. Identificador: %s | "
                "Estado anterior: %s | Ação: %s | Próximo estado: %s | "
                "Terminal: %s",
                resultado.identificador,
                resultado.estado_anterior,
                resultado.acao,
                resultado.proximo_estado,
                resultado.terminal,
            )
        except TransicaoInvalidaError as exc:
            estado_atual = dados_transicao.estado_atual
            resultado = ResultadoWorkflow(
                identificador=contexto.identificador,
                estado_anterior=estado_atual,
                acao=dados_transicao.acao.nome,
                proximo_estado=estado_atual,
                terminal=False,
                sucesso=False,
                inicio=dados_transicao.inicio_transicao,
                fim=datetime.now(UTC),
                autor=contexto.autor,
                contexto=contexto,
                mensagem=f"Transição não executada! {str(exc)}",
            )
            logger.warning(
                "Falha na execução do workflow. Identificador: %s | "
                "Estado atual: %s | Ação: %s | Motivo: %s",
                contexto.identificador,
                estado_atual,
                dados_transicao.acao.nome,
                exc,
            )

        return resultado.para_dict()
